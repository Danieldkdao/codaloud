cmake_minimum_required(VERSION 3.22)
project(codaloud_workspace LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(USE_SSH OFF CACHE BOOL "" FORCE)
set(USE_NTLMCLIENT OFF CACHE BOOL "" FORCE)
set(REGEX_BACKEND builtin CACHE STRING "" FORCE)
if(APPLE)
  set(USE_HTTPS SecureTransport CACHE STRING "" FORCE)
else()
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/mbedtls EXCLUDE_FROM_ALL)
  set(MBEDTLS_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/mbedtls/include" CACHE PATH "" FORCE)
  set(MBEDTLS_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/vendor/mbedtls/library/libmbedtls.a" CACHE FILEPATH "" FORCE)
  set(MBEDX509_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/vendor/mbedtls/library/libmbedx509.a" CACHE FILEPATH "" FORCE)
  set(MBEDCRYPTO_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/vendor/mbedtls/library/libmbedcrypto.a" CACHE FILEPATH "" FORCE)
  set(USE_HTTPS mbedTLS CACHE STRING "" FORCE)
  set(CERT_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/cpp/system-trust.txt" CACHE PATH "" FORCE)
endif()
add_subdirectory(vendor/libgit2 EXCLUDE_FROM_ALL)
add_library(workspace-core STATIC cpp/workspace.cpp cpp/files.cpp cpp/git-core.cpp cpp/git-diff.cpp cpp/git-history.cpp cpp/git-mutations.cpp)
target_include_directories(workspace-core PUBLIC cpp vendor vendor/libgit2/include)
target_link_libraries(workspace-core PUBLIC libgit2package)
if(NOT APPLE)
  target_link_libraries(workspace-core PUBLIC mbedtls mbedx509 mbedcrypto)
endif()
if(ANDROID)
  add_library(codaloud-workspace SHARED cpp/android-bridge.cpp)
  target_link_libraries(codaloud-workspace PRIVATE workspace-core log)
  target_link_options(codaloud-workspace PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(workspace-cli cpp/workspace-cli.cpp)
  target_link_libraries(workspace-cli PRIVATE workspace-core)
endif()
