cmake_minimum_required(VERSION 3.22)
project(codaloud_workspace LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(USE_SSH OFF CACHE BOOL "" FORCE)
set(USE_NTLMCLIENT OFF CACHE BOOL "" FORCE)
set(REGEX_BACKEND builtin CACHE STRING "" FORCE)
if(APPLE)
  set(USE_HTTPS SecureTransport CACHE STRING "" FORCE)
endif()
add_subdirectory(vendor/libgit2 EXCLUDE_FROM_ALL)
add_library(workspace-core STATIC cpp/workspace.cpp cpp/files.cpp cpp/git-core.cpp)
target_include_directories(workspace-core PUBLIC cpp vendor vendor/libgit2/include)
target_link_libraries(workspace-core PUBLIC libgit2package)
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(workspace-cli cpp/workspace-cli.cpp)
  target_link_libraries(workspace-cli PRIVATE workspace-core)
endif()
